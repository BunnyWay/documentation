The Bunny Player is an iframe you control through [player.js](https://github.com/embedly/player.js), a small library that wraps the `postMessage` protocol the player speaks. This guide builds a `BunnyPlayer` component that renders the iframe, emits player events, and exposes the player instance for controlling playback.

The component is written so that it also works under Nuxt server rendering: player.js is imported in `onMounted`, because it reads `window` when imported, and the iframe is rendered only after the library is available, because a `Player` must exist before its iframe finishes loading.

  <Step title="Create the component">
    ```vue components/BunnyPlayer.vue
    <script setup lang="ts">
    import { computed, onMounted, ref, shallowRef, watch } from "vue";
    import type { Player, TimeUpdate } from "player.js";

    type PlayerJs = (typeof import("player.js"))["default"];

    const props = withDefaults(
      defineProps<{
        libraryId: string;
        videoId: string;
        /** Player parameters such as autoplay, muted, captions, or t. */
        params?: Record<string, string | number | boolean>;
        title?: string;
      }>(),
      { title: "Video player" },
    );

    const emit = defineEmits<{
      ready: [player: Player];
      play: [];
      pause: [];
      ended: [];
      timeupdate: [time: TimeUpdate];
    }>();

    const iframe = ref<HTMLIFrameElement | null>(null);
    const playerjs = shallowRef<PlayerJs | null>(null);
    const player = shallowRef<Player | null>(null);

    const src = computed(() => {
      const query = new URLSearchParams(
        Object.entries(props.params ?? {}).map(([key, value]) => [key, String(value)]),
      ).toString();
      return `https://player.mediadelivery.net/embed/${props.libraryId}/${props.videoId}${query ? `?${query}` : ""}`;
    });

    onMounted(async () => {
      // player.js reads window when imported, so load it in the browser only.
      playerjs.value = (await import("player.js")).default;
    });

    // The iframe is keyed by src, so this runs for every new iframe element,
    // right after it is inserted and before it has finished loading.
    watch(
      iframe,
      (el, _previous, onCleanup) => {
        if (!el || !playerjs.value) return;

        // player.js has no teardown API. This flag stops stale listeners
        // from firing after the video changes or the component unmounts.
        let active = true;
        const instance = new playerjs.value.Player(el);

        instance.on("ready", () => active && emit("ready", instance));
        instance.on("play", () => active && emit("play"));
        instance.on("pause", () => active && emit("pause"));
        instance.on("ended", () => active && emit("ended"));
        instance.on("timeupdate", (time) => active && emit("timeupdate", time));

        player.value = instance;
        onCleanup(() => {
          active = false;
          player.value = null;
        });
      },
      { flush: "post" },
    );

    defineExpose({ player });
    </script>

    <template>
      <iframe
        v-if="playerjs"
        ref="iframe"
        :key="src"
        :src="src"
        :title="title"
        class="bunny-player"
        allow="autoplay; encrypted-media; picture-in-picture; fullscreen"
        allowfullscreen
      />
      <!-- Hold the space until player.js is loaded so the iframe cannot
           finish loading before the Player exists. -->
      <div v-else class="bunny-player" aria-hidden="true" />
    </template>

    <style scoped>
    .bunny-player {
      display: block;
      width: 100%;
      height: auto;
      aspect-ratio: 16 / 9;
      border: 0;
      background: #000;
    }
    </style>
    ```

    In Nuxt, save the file under `components/` and it is auto-imported. No `<ClientOnly>` wrapper is needed: the server renders the placeholder, and the browser swaps in the iframe.
  </Step>
  <Step title="Render a video">
    Pass the library ID and video GUID. Both are shown on the video's page in the dashboard, and in the embed code it generates.

    ```vue
    <script setup lang="ts">
    import BunnyPlayer from "./components/BunnyPlayer.vue";
    </script>

    <template>
      <BunnyPlayer
        library-id="12345"
        video-id="your-video-guid"
        :params="{ autoplay: false, preload: true }"
        @ended="console.log('Video finished')"
      />
    </template>
    ```

    `params` accepts any of the [player parameters](/stream/embedding#supported-parameters), such as `captions`, `t`, or `muted`.
  </Step>

## Control playback

The `ready` event carries the `Player` instance. Keep it in a `shallowRef`, so Vue does not wrap the instance in a deep reactive proxy, and call its methods from your own controls:

```vue
<script setup lang="ts">
import { ref, shallowRef } from "vue";
import type { Player } from "player.js";
import BunnyPlayer from "./components/BunnyPlayer.vue";

const player = shallowRef<Player | null>(null);
const playing = ref(false);

function toggle() {
  playing.value ? player.value?.pause() : player.value?.play();
}
</script>

<template>
  <BunnyPlayer
    library-id="12345"
    video-id="your-video-guid"
    @ready="player = $event"
    @play="playing = true"
    @pause="playing = false"
  />
  <button @click="toggle">{{ playing ? "Pause" : "Play" }}</button>
  <button @click="player?.setCurrentTime(0)">Restart</button>
  <button @click="player?.mute()">Mute</button>
</template>
```

Getters are asynchronous because the answer comes back from the iframe. They take a callback instead of returning a value:

```ts
player.value?.getCurrentTime((seconds) => console.log(seconds));
player.value?.getDuration((seconds) => console.log(seconds));
```

Playback speed is not a named method in player.js. Send the command directly, as described in [Playback rate](/stream/playback-api#playback-rate):

```ts
player.value?.send({ method: "setPlaybackRate", value: 1.5 });
```

Browsers block unmuted `play()` calls until the viewer has interacted with the page. Call `mute()` first if playback must start without a click. See [Playback control API](/stream/playback-api) for every method and event.

## Track progress

`timeupdate` fires several times a second with the current position and duration. Throttle it before writing to your backend:

```vue
<script setup lang="ts">
import BunnyPlayer from "./components/BunnyPlayer.vue";

const props = defineProps<{ videoId: string; resumeAt?: number }>();
let lastSaved = 0;

function onTimeUpdate({ seconds, duration }: { seconds: number; duration: number }) {
  if (seconds - lastSaved < 5) return;
  lastSaved = seconds;
  saveProgress(props.videoId, seconds, duration);
}
</script>

<template>
  <BunnyPlayer
    library-id="12345"
    :video-id="videoId"
    :params="resumeAt ? { t: resumeAt } : undefined"
    @timeupdate="onTimeUpdate"
    @ended="markComplete(videoId)"
  />
</template>
```

Passing the saved position back as the `t` parameter resumes playback from where the viewer left off.

## Multiple players on one page

player.js matches the `ready` message to an iframe by its `src`. Two iframes embedding the same video with identical URLs confuse it. Add a unique parameter to each so every `src` differs, for example `:params="{ instance: useId() }"`. The player ignores parameters it does not recognise.

## Load player.js from the CDN instead

To skip the npm dependency, load the hosted script and read `window.playerjs`. In Nuxt, add it to `nuxt.config.ts`:

```ts nuxt.config.ts
export default defineNuxtConfig({
  app: {
    head: {
      script: [{ src: "https://assets.mediadelivery.net/playerjs/playerjs-latest.min.js" }],
    },
  },
});
```

In a plain Vue app, add the `<script>` tag to `index.html`. Then replace the dynamic import in `onMounted` with `playerjs.value = window.playerjs`, and declare the global:

```ts
declare global {
  interface Window {
    playerjs: (typeof import("player.js"))["default"];
  }
}
```

## Troubleshooting

<AccordionGroup>
  <Accordion title="ReferenceError: window is not defined">
    player.js is being imported during server rendering. Keep the `import("player.js")` inside `onMounted`, as the component above does. A static `import playerjs from "player.js"` at the top of `<script setup>` runs on the server in Nuxt.
  </Accordion>
  <Accordion title="The ready event never fires">
    The iframe finished loading before the `Player` was created. Render the iframe only after player.js has loaded, as the component above does, so the two happen in the same update.

    The player also refuses messages from a page whose origin does not match the iframe's `Referer`. If your site sets `Referrer-Policy: no-referrer`, add `referrerpolicy="strict-origin-when-cross-origin"` to the iframe. See [Requirements](/stream/playback-api#requirements).
  </Accordion>
  <Accordion title="Events fire twice after changing the video">
    Each `Player` registers a global `message` listener that is never removed. Guard callbacks with a flag that the watcher cleanup flips to `false`, as in the component above, so only the current instance forwards events.
  </Accordion>
  <Accordion title="The iframe shows a 403">
    The library's allowed domains, direct access block, or token authentication is rejecting the embed. Check the [embedding restrictions](/stream/embedding#embedding-restrictions) for each setting.
  </Accordion>
</AccordionGroup>
